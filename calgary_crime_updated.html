# Forecast with trend ans Sazonality
from statsmodels.tsa.deterministic import DeterministicProcess
from statsmodels.api import OLS

# create the deterministic process

def forecast_linear_trend(train, steps):
    # extract date index and create trend and sazonality componests
    dates = train.index
    dp = DeterministicProcess(index = dates,
                               constant=True, 
                               order=1, 
                               seasonal=True, 
                               drop = True)
    X = dp.in_sample()
    y = train['Crime Count']

    #fit the model
    model = OLS(y, X).fit()

    #generate forecast
    X_fore = dp.out_of_sample(steps=steps)
    forecast = model.predict(X_fore)
    return forecast

def forecast_b(train, steps):

    y = train['Crime Count']
    # Fit the model
    model = OLS(y, X).fit()
    # Generate forecast
    X_fore = dp.out_of_sample(steps=steps)
    forecast = model.predict(X_fore)
    return forecast

# model C - Arima
from statsmodels.tsa.arima.model import ARIMA

def forecast_arima(train, steps, order=(1,1,1), seasonal_order=(1, 1, 1, 12)):
    # fit the model
    model = ARIMA(train['Crime Count'], order=order, seasonal_order=seasonal_order)
    model_fit = model.fit()

    # generate forecast
    forecast = model_fit.forecast(steps=steps)
    return forecast

from sklearn.metrics import mean_absolute_error, mean_squared_error, mean_absolute_percentage_error
# calculate the error metrics
def calculate_error(test, forecast):
    mae = mean_absolute_error(test, forecast).round(0)
    mse = mean_squared_error(test, forecast).round(0)
    mape = mean_absolute_percentage_error(test, forecast).round(0)
    return mae, mse, mape
